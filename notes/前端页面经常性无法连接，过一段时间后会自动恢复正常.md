# 前端页面经常性无法连接，过一段时间后会自动恢复正常



# 适配版本


v2.0R25c21

v2.0R25c22


# 问题现象


前端页面经常性无法连接，过一段时间后会自动恢复正常


# 解决步骤


在Linux中查看当前网关IP并配置网关静态ARP，可以按照以下步骤操作：


## 1. 查看当前网关IP



### 方法1：使用ip route命令


ip route show default
# 或者
ip route | grep default
# 或者
ip route list 0/0


### 方法2：使用route命令


route -n


### 方法3：使用netstat命令


netstat -rn


### 方法4：查看特定接口的网关


# 查看所有接口的网关信息
ip route show

# 查看eth0接口的网关
ip route show dev eth0


## 2. 获取网关的MAC地址


在配置静态ARP之前，需要先获取网关的MAC地址：

# 先ping一下网关确保ARP表中有记录
ping -c 1 192.168.1.1

# 查看网关的MAC地址
arp -n 192.168.1.1
# 或者
ip neigh show 192.168.1.1


## 3. 配置网关静态ARP



### 方法1：使用arp命令（临时）


# 假设网关IP是192.168.1.1，MAC是00:11:22:33:44:55
sudo arp -s 192.168.1.1 00:11:22:33:44:55


### 方法2：使用ip neigh命令（临时）


sudo ip neigh add 192.168.1.1 lladdr 00:11:22:33:44:55 dev eth0 nud permanent


### 方法3：创建开机自动执行脚本（永久）


创建systemd服务：

# 创建服务文件
sudo nano /etc/systemd/system/gateway-arp.service

添加以下内容：

[Unit]
Description=Static ARP for Gateway
After=network.target
Wants=network.target

[Service]
Type=oneshot
ExecStart=/usr/local/bin/setup-gateway-arp.sh
RemainAfterExit=yes

[Install]
WantedBy=multi-user.target

创建ARP设置脚本：

sudo nano /usr/local/bin/setup-gateway-arp.sh

添加内容：

#!/bin/bash

# 等待网络就绪
sleep 5

# 获取默认网关IP
GATEWAY_IP=$(ip route show default | awk '/default/ {print $3}')
INTERFACE=$(ip route show default | awk '/default/ {print $5}')

# 如果获取到网关IP
if [ ! -z "$GATEWAY_IP" ] && [ ! -z "$INTERFACE" ]; then
    echo "Setting static ARP for gateway $GATEWAY_IP on interface $INTERFACE"
    
    # 先ping一次获取MAC地址
    ping -c 1 $GATEWAY_IP > /dev/null 2>&1
    
    # 获取MAC地址
    GATEWAY_MAC=$(arp -n $GATEWAY_IP | awk '/'$GATEWAY_IP'/ {print $3}')
    
    # 如果成功获取MAC地址，设置静态ARP
    if [ ! -z "$GATEWAY_MAC" ] && [ "$GATEWAY_MAC" != "(incomplete)" ]; then
        ip neigh replace $GATEWAY_IP lladdr $GATEWAY_MAC dev $INTERFACE nud permanent
        echo "Static ARP set: $GATEWAY_IP -> $GATEWAY_MAC"
    else
        echo "Failed to get MAC address for gateway $GATEWAY_IP"
    fi
else
    echo "Failed to get gateway information"
fi

设置权限并启用服务：

sudo chmod +x /usr/local/bin/setup-gateway-arp.sh
sudo systemctl enable gateway-arp.service
sudo systemctl start gateway-arp.service


### 方法4：手动配置永久静态ARP


如果知道确切的网关IP和MAC地址，可以直接配置：

# 编辑rc.local
sudo nano /etc/rc.local

# 添加以下内容（在exit 0之前）
arp -s 192.168.1.1 00:11:22:33:44:55

或者创建cron任务：

# 编辑root用户的crontab
sudo crontab -e

# 添加在启动时执行
@reboot sleep 30 && arp -s 192.168.1.1 00:11:22:33:44:55


## 4. 验证配置



### 检查静态ARP条目


# 查看特定网关的ARP条目
arp -n 192.168.1.1

# 或者查看所有ARP条目
arp -a

# 使用ip命令查看
ip neigh show 192.168.1.1


### 测试网络连接


# 测试到网关的连接
ping -c 3 192.168.1.1

# 测试外网连接
ping -c 3 8.8.8.8


## 5. 完整的自动化脚本示例


这里是一个更完整的自动化脚本：

#!/bin/bash
# setup-gateway-arp.sh

setup_gateway_arp() {
    # 等待网络稳定
    sleep 10
    
    # 获取网关信息
    GATEWAY_IP=$(ip route show default | awk '/default/ {print $3}')
    INTERFACE=$(ip route show default | awk '/default/ {print $5}')
    
    if [ -z "$GATEWAY_IP" ] || [ -z "$INTERFACE" ]; then
        echo "Error: Cannot determine gateway or interface"
        return 1
    fi
    
    echo "Gateway: $GATEWAY_IP"
    echo "Interface: $INTERFACE"
    
    # 获取当前ARP条目
    CURRENT_MAC=$(ip neigh show $GATEWAY_IP | awk '/'$GATEWAY_IP'/ {print $5}')
    
    if [ -z "$CURRENT_MAC" ] || [ "$CURRENT_MAC" = "FAILED" ]; then
        echo "Pinging gateway to populate ARP table..."
        ping -c 2 $GATEWAY_IP > /dev/null 2>&1
        sleep 2
        CURRENT_MAC=$(ip neigh show $GATEWAY_IP | awk '/'$GATEWAY_IP'/ {print $5}')
    fi
    
    if [ -n "$CURRENT_MAC" ] && [ "$CURRENT_MAC" != "FAILED" ]; then
        echo "Setting static ARP: $GATEWAY_IP -> $CURRENT_MAC"
        ip neigh replace $GATEWAY_IP lladdr $CURRENT_MAC dev $INTERFACE nud permanent
        echo "Static ARP configured successfully"
    else
        echo "Error: Cannot determine gateway MAC address"
        return 1
    fi
}

setup_gateway_arp

这样配置后，每次系统启动时都会自动为网关设置静态ARP条目，确保网关的MAC地址不会被ARP欺骗攻击篡改。
